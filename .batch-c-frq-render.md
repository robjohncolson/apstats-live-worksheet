      function handleHelperClick(kind, id, unit, data, questionId) {
        if (!data.frqHelpers) data.frqHelpers = defaultFrqHelpers();
        if (!data.frqHelpers.confirmed) {
          showFrqHelperModal(() => {
            data.frqHelpers.confirmed = true;
            saveSoon();
            doHelperAction(kind, id, unit, data, questionId);
          });
          return;
        }
        doHelperAction(kind, id, unit, data, questionId);
      }

      function doHelperAction(kind, id, unit, data, questionId) {
        if (kind === 'formula') {
          recordHelperUsed(unit, 'formula', id, state);
          renderRemediation();
          renderActiveProbe();
          saveSoon();
        } else if (kind === 'drill') {
          data.frqHelpers.activeDrillQuestionId = id;
          renderRemediation();
          saveSoon();
        }
      }

      function showFrqHelperModal(onConfirm) {
        const backdrop = make('div', 'sg-modal-backdrop sg-frq-helper-modal');
        const modal = make('div', 'sg-modal');
        modal.setAttribute('role', 'dialog');
        modal.setAttribute('aria-modal', 'true');
        modal.tabIndex = -1;
        modal.appendChild(make('h2', '', 'Using helpers will reduce your score.'));
        modal.appendChild(make('p', 'sg-remediation-body', 'Continue?'));
        const btnRow = make('div', 'sg-next-row');
        const confirmBtn = make('button', '', 'Yes, show helper');
        confirmBtn.type = 'button';
        const escHandler = e => {
          if (e.key === 'Escape') {
            backdrop.remove();
            document.removeEventListener('keydown', escHandler);
          }
        };
        confirmBtn.addEventListener('click', () => {
          backdrop.remove();
          document.removeEventListener('keydown', escHandler);
          onConfirm();
        });
        const cancelBtn = make('button', '', 'Cancel');
        cancelBtn.type = 'button';
        cancelBtn.addEventListener('click', () => {
          backdrop.remove();
          document.removeEventListener('keydown', escHandler);
        });
        btnRow.appendChild(confirmBtn);
        btnRow.appendChild(cancelBtn);
        modal.appendChild(btnRow);
        backdrop.appendChild(modal);
        document.addEventListener('keydown', escHandler);
        backdrop.addEventListener('click', e => {
          if (e.target === backdrop) {
            backdrop.remove();
            document.removeEventListener('keydown', escHandler);
          }
        });
        document.body.appendChild(backdrop);
        modal.focus();
      }
