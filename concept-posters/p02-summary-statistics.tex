\PosterZone{16.23in}{19.64in}{ZONE B · WHAT THE SYMBOLS MEAN}{%
  \begin{minipage}[t]{9.65in}
    \MiniHeading{SAMPLE}\par\vspace{.08in}
    {\fontsize{29pt}{37pt}\selectfont
      \(\bar{x}\) = sample mean \qquad \(s\) = sample SD\par
      five-number summary: min, \(Q_1\), median, \(Q_3\), max}
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{9.65in}
    \MiniHeading{POPULATION + RESISTANCE}\par\vspace{.08in}
    {\fontsize{29pt}{37pt}\selectfont
      \(\mu\) = population mean \qquad \(\sigma\) = population SD\par
      \textbf{resistant:} median, IQR \quad \textbf{not resistant:} mean, SD, range}
  \end{minipage}%
}
