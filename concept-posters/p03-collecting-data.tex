\PosterZone{4.72in}{16.4in}{ZONE A \textbullet\ HOW THE DATA ARE COLLECTED}{%
  \FormulaCardSized{39pt}{\text{Data from random sample or randomized experiment}}
  \vspace{.23in}
  \begin{minipage}[t]{6.35in}
    \vspace{0pt}
    \TextCard{%
      \MiniHeading{SAMPLING METHODS}\par\vspace{.15in}
      {\fontsize{27pt}{35pt}\selectfont
      \textbf{SRS:} every possible sample of size \(n\) has the same chance.\par\vspace{.18in}
      \textbf{Stratified:} divide into similar groups; randomly sample within each.\par\vspace{.18in}
      \textbf{Cluster:} randomly choose whole groups; include everyone in them.\par\vspace{.18in}
      \textbf{Systematic:} start randomly, then select every \(k\)th individual.}
    }
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{6.35in}
    \vspace{0pt}
    \TextCard{%
      \MiniHeading{SOURCES OF BIAS}\par\vspace{.15in}
      {\fontsize{27pt}{35pt}\selectfont
      \textbf{Undercoverage:} some groups are left out or hard to reach.\par\vspace{.15in}
      \textbf{Nonresponse:} selected people do not respond.\par\vspace{.15in}
      \textbf{Response bias:} wording or reporting distorts answers.\par\vspace{.15in}
      \textbf{Voluntary response:} people choose to join the sample.\par\vspace{.15in}
      \textbf{Convenience:} choose people who are easiest to reach.}
    }
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{6.65in}
    \vspace{0pt}
    \TextCard{%
      \MiniHeading{EXPERIMENTS}\par\vspace{.12in}
      {\fontsize{25pt}{32pt}\selectfont
      \textbf{Control:} hold other conditions alike; compare treatments.\par
      \textbf{Random assignment:} use chance to form treatment groups.\par
      \textbf{Replication:} use enough experimental units.\par\vspace{.14in}
      \textbf{Completely randomized:} assign all units at random.\par
      \textbf{Randomized block:} group similar units, then randomize within.\par
      \textbf{Matched pairs:} compare two related measurements or units.\par\vspace{.14in}
      \textbf{Placebo:} an inactive treatment.\par
      \textbf{Blinding:} hide treatment identity from subjects or evaluators.\par
      \textbf{Confounding:} effects are mixed and cannot be separated.}
    }
  \end{minipage}%
}

  \begin{minipage}[t]{6.65in}
