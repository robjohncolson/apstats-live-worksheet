\begin{callout}{\IconBook\ DESCRIBING A DISTRIBUTION (keep on the page)}{calloutgreen}
\textbf{Shape} (symmetric / skewed left / skewed right; one peak or two) \textperiodcentered\
\textbf{Center} (median or mean) \textperiodcentered\ \textbf{Variability} (range, IQR)
\textperiodcentered\ \textbf{Unusual features} (gaps, outliers, clusters) --- always \textbf{in context, with units}.\\
A long tail pulls the \textbf{mean} toward it; the \textbf{median} resists.
``Skewed right'' means the \emph{tail} points right, not the peak.
\end{callout}
