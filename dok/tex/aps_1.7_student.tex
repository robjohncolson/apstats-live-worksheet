\begin{callout}{\IconBook\ CENTER, SPREAD, AND OUTLIERS (keep on the page)}{calloutgreen}
$\bar{x}=\dfrac{\sum x_i}{n}$ \textperiodcentered\
$s_x=\sqrt{\dfrac{\sum(x_i-\bar{x})^2}{n-1}}$ \textperiodcentered\
$\mathrm{IQR}=Q_3-Q_1$. For odd $n$, use the ordered values \textbf{before the overall median's position} for $Q_1$
and those \textbf{after its position} for $Q_3$; do not include the overall median itself in either half.\\
Outlier fences: $Q_1-1.5(\mathrm{IQR})$ and $Q_3+1.5(\mathrm{IQR})$. Mean and SD are
\textbf{nonresistant}; median and IQR are \textbf{resistant}. Match the pair to the distribution.
\end{callout}
