\textbf{1} & 0 of 0--9 & 5 digits; $2+$ L \\ \hline
\textbf{2} & 00--11 of 00--99 & 5 pairs; exactly 2 L \\ \hline
\end{tabular}
\end{center}

\begin{callout}{\IconBook\ THREE MATCHES MAKE A VALID SIMULATION (keep on the page)}{calloutgreen}
\textbf{Chance assignment:} the fraction of equally likely labels for each outcome must match its real probability.\\
\textbf{One trial:} reproduce the stated number of independent selections.\\
\textbf{Success event:} count exactly the outcomes named in the question; ``at least 2'' includes 2, 3, 4, and 5.
Repeat many trials and estimate probability with $\dfrac{\text{successful trials}}{\text{total trials}}$.
\end{callout}
